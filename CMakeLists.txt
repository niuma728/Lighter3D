cmake_minimum_required(VERSION 3.24)
project(Lighter3D VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)

include(CTest)
add_library(lighter_logic src/Lighter.cpp src/DemoController.cpp)
target_include_directories(lighter_logic PUBLIC include)
if(BUILD_TESTING)
    add_executable(LighterStateTests tests/LighterStateTests.cpp)
    target_link_libraries(LighterStateTests PRIVATE lighter_logic)
    add_test(NAME lighter_state COMMAND LighterStateTests)
endif()

option(LIGHTER_BUILD_APP "Build the OpenGL application" ON)
if(NOT LIGHTER_BUILD_APP)
    return()
endif()

include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(USE_MSVC_RUNTIME_LIBRARY_DLL OFF CACHE BOOL "" FORCE)
set(GLM_BUILD_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz
    URL_HASH SHA256=c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01)
FetchContent_Declare(glm URL https://github.com/g-truc/glm/archive/refs/tags/1.0.1.tar.gz
    URL_HASH SHA256=9f3174561fd26904b23f0db5e560971cbf9b3cbda0b280f04d5c379d03bf234c)
FetchContent_Declare(glad URL https://github.com/Dav1dde/glad/archive/refs/tags/v2.0.8.tar.gz
    URL_HASH SHA256=44f06f9195427c7017f5028d0894f57eb216b0a8f7c4eda7ce883732aeb2d0fc SOURCE_SUBDIR cmake)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_OBJ_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_NO_EXPORT ON CACHE BOOL "" FORCE)
FetchContent_Declare(assimp URL https://github.com/assimp/assimp/archive/refs/tags/v5.4.3.tar.gz
    URL_HASH SHA256=66dfbaee288f2bc43172440a55d0235dfc7bf885dda6435c038e8000e79582cb)
FetchContent_MakeAvailable(glfw glm glad assimp)
glad_add_library(glad_gl33 STATIC REPRODUCIBLE API gl:core=3.3 EXTENSIONS NONE)

add_library(lighter_effects src/ParticleSystem.cpp src/Simulation.cpp)
target_link_libraries(lighter_effects PUBLIC lighter_logic glm::glm)
if(BUILD_TESTING)
    add_executable(EffectsTests tests/EffectsTests.cpp src/Camera.cpp)
    target_link_libraries(EffectsTests PRIVATE lighter_effects)
    add_test(NAME effects_simulation_camera COMMAND EffectsTests)
    if(MINGW)
        target_link_options(EffectsTests PRIVATE -static)
    endif()
endif()
add_executable(Lighter3D src/main.cpp src/Camera.cpp src/Shader.cpp src/Model.cpp src/Renderer.cpp src/EffectsRenderer.cpp src/Hud.cpp)
target_include_directories(Lighter3D PRIVATE include)
target_link_libraries(Lighter3D PRIVATE lighter_effects glfw glad_gl33 assimp::assimp)
target_compile_definitions(Lighter3D PRIVATE GLFW_INCLUDE_NONE NOMINMAX)
if(MSVC)
    target_compile_options(Lighter3D PRIVATE /W4 /utf-8)
    target_compile_options(lighter_logic PRIVATE /W4 /utf-8)
    target_compile_options(lighter_effects PRIVATE /W4 /utf-8)
else()
    target_compile_options(Lighter3D PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(lighter_logic PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(lighter_effects PRIVATE -Wall -Wextra -Wpedantic)
    if(MINGW)
        target_link_options(Lighter3D PRIVATE -static)
        if(BUILD_TESTING)
            target_link_options(LighterStateTests PRIVATE -static)
        endif()
    endif()
endif()
set_target_properties(Lighter3D PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_target(lighter_resources ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets/models" "$<TARGET_FILE_DIR:Lighter3D>/assets/models"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:Lighter3D>/shaders"
    VERBATIM)
add_dependencies(Lighter3D lighter_resources)
install(TARGETS Lighter3D RUNTIME DESTINATION . COMPONENT Runtime)
install(DIRECTORY assets/models DESTINATION assets COMPONENT Runtime)
install(DIRECTORY shaders DESTINATION . COMPONENT Runtime)
install(FILES README.md DESTINATION . COMPONENT Runtime)
install(FILES docs/development.md DESTINATION docs COMPONENT Runtime)
install(FILES docs/demo.md DESTINATION docs COMPONENT Runtime)
install(FILES "Lighter3D_设计文档.md" DESTINATION docs RENAME design.md COMPONENT Runtime)
install(DIRECTORY docs/screenshots DESTINATION docs COMPONENT Runtime)
install(DIRECTORY docs/validation DESTINATION docs COMPONENT Runtime)
install(DIRECTORY docs/licenses DESTINATION . COMPONENT Runtime)
